\section{Operational definitions}

\begin{definition}[Spacetime fault; weight]\label{def:fault}
For a quantum memory experiment with $n_d$ data qubits and $R$ syndrome
rounds, a \emph{spacetime fault set} $F$ is a set of elementary faults:
(i)~a Pauli operator applied to one data qubit at one inter-round
location; (ii)~one faulty syndrome (ancilla) measurement outcome. The
\emph{weight} $|F|$ counts elementary faults. A fault set is
\emph{adversarially placed} if its support is chosen arbitrarily in space
and time.
\end{definition}

\begin{definition}[Injection-budget adversary $\mathfrak{A}_1^{\mathrm{st}}(W)$]
\label{def:adversary}
The adversary selects any fault set $F$ of injected data-qubit Pauli
faults with $|F|\le W$, with full knowledge of the code, the schedule,
and the decoder, \emph{without} access to syndrome outcomes before
selection (non-adaptive) and without any influence on native hardware
noise $N$ or on the decoding procedure. The native noise contributes an
independent fault set $N$ drawn from the device distribution
$\mathcal{P}_{\mathrm{dev}}$.
\end{definition}

\begin{definition}[Operational ACE value]\label{def:acevalue}
For implemented memory $\mathcal{M}$ (code $+$ schedule $+$ decoder),
\[
\lambda_{\mathcal{M}}(W)\;:=\;\max_{F:\,|F|\le W}\;
\Pr_{N\sim\mathcal{P}_{\mathrm{dev}}}\!\big[\text{logical failure}\mid F,N\big],
\qquad V_{\mathcal{M}}(W):=1-\lambda_{\mathcal{M}}(W).
\]
$\lambda_{\mathcal{M}}(W)$ is directly measurable: inject $F$ on the
device, decode, count logical failures.
\end{definition}

\section{The spacetime-budget theorem}

\begin{theorem}[Bounded-weight correctability and sharpness]\label{thm:spacetime}
Let $\mathcal{M}$ be the distance-$d$ ($d$ odd) repetition memory with
$R\ge 1$ rounds, decoded by minimum-weight matching on the spacetime
detector graph, and let $t=\lfloor(d-1)/2\rfloor$.
\begin{enumerate}[label=(\alph*),leftmargin=1.6em]
\item \emph{Ideal extraction.} If $N=\varnothing$, then
$\lambda_{\mathcal{M}}(W)=0$ for all $W\le t$: every adversarially
placed injection of weight at most $t$ is corrected.
\item \emph{Sharpness.} There exists an explicitly constructible
injection $F$ with $|F|=t+1$ such that minimum-weight matching fails
with probability one: $\lambda_{\mathcal{M}}(t+1)=1$ already at
$N=\varnothing$.
\item \emph{Composed, testable bound.} For arbitrary native noise,
\[
\lambda_{\mathcal{M}}(W)\;\le\;\Pr_{N}\!\big[\,|N|\ge (t+1)-W\,\big]
\qquad(0\le W\le t),
\]
and in particular
$\lambda_{\mathcal{M}}(0)=\Pr[|N|\ge t+1]$ is the baseline logical
failure rate and $\lambda_{\mathcal{M}}(t)\le\Pr[|N|\ge 1]$.
\end{enumerate}
\end{theorem}

\begin{proof}
(a)~In the spacetime detector graph of the repetition memory, elementary
faults are edges: data faults are spatial edges (connecting
neighbouring-space detectors in one round, or one detector to a spatial
boundary) and measurement faults are temporal edges (same detector,
consecutive rounds). Any logical operator chain must connect the two
spatial boundaries and therefore contains at least $d$ spatial edges;
temporal edges do not advance the spatial coordinate, so every logical
chain has weight $\ge d$. A fault set $F$ flips the set of detectors at
the boundary of its chain; the decoder returns a minimum-weight chain
$C$ with the same boundary, $|C|\le|F|$. Failure requires $F\oplus C$
to contain a logical chain, hence
$|F|+|C|\ge|F\oplus C|\ge d$, so $2|F|\ge d$, i.e.\
$|F|\ge t+1$. Contrapositive: $|F|\le t$ implies success.
(b)~Take $X$ on $t{+}1$ consecutive data qubits centred in the line,
injected in one round. The syndrome has two detectors separated by a
path of weight $t{+}1$ through the injected string, while the two
boundary paths have total weight $d-(t{+}1)=t$. Minimum-weight matching
chooses the boundary completion of weight $t$, and $F\oplus C$ is the
full logical string; failure is certain. (c)~With $E=N\cup F$ the same
chain argument gives failure $\Rightarrow |N|+|F|\ge t+1$, hence
$|N|\ge t+1-W$; take probabilities.
\end{proof}

\begin{remark}[Scope]
Theorem~\ref{thm:spacetime} is stated for repetition memories, whose
spacetime graph is a chain; the proof uses only the chain complex and
minimum-weight decoding, so (a) extends to any CSS memory whose
spacetime distance is $d$ under the same decoder class. For BB72 the
analogous single-round statement at $W\le t=2$ is established
\emph{exhaustively} by the finite-instance certificate
(Sec.~Certificate of the main text): every weight-$\le2$ Pauli pattern
is corrected. We therefore regard the repetition-code theorem as the
proved spine and BB72 as the certificate-supported high-rate instance;
a general spacetime-distance proof for BB memories is future work.
\end{remark}

\section{A measurable bridge for the diamond-norm gap}

\begin{proposition}[Pauli-channel diamond bridge]\label{prop:bridge}
Let $\Lambda$ be a Pauli channel on one logical qubit with distribution
$\{p_P\}_{P\in\{I,X,Y,Z\}}$. Then
$\tfrac12\|\Lambda-\mathrm{id}\|_\diamond = 1-p_I$.
Consequently, if the effective logical channel of the implemented
memory is Pauli (e.g.\ after Pauli twirling of the recovery), the
logical diamond-norm error satisfies
\[
\varepsilon_{\mathrm{FT}}^{\mathrm{logical}}
= 2\bigl(1-p_I\bigr)
\le 2\max_{\beta\in\{X,Z\}}\mathrm{LFR}_{\beta},
\]
where $\mathrm{LFR}_\beta$ is the logical failure rate of the
$\beta$-basis memory experiment.
\end{proposition}
\begin{proof}
Pauli channels are covariant; their diamond norm is attained on a
maximally entangled input and equals the $\ell_1$ norm of the
difference of the Pauli distributions (Watrous, \emph{Theory of Quantum
Information}, semidefinite-program characterization of the diamond norm
for Pauli channels). For $\Lambda$ versus $\mathrm{id}$:
$\sum_P|p_P-\delta_{P,I}|=(1-p_I)+(1-p_I)=2(1-p_I)$. Under the Pauli
assumption, $1-p_I$ is the total logical Pauli probability, bounded by
the worse of the two basis experiments.
\end{proof}

\begin{remark}[What this buys]
Proposition~\ref{prop:bridge} upgrades the decoder-dependent ACE gap
(Remark on decoder dependence in the main text) from an LFR
\emph{proxy} to a \emph{bound}: under the Pauli approximation for the
logical channel, $\varepsilon_{\mathrm{FT}}^{\mathrm{logical}}\le
2\,\mathrm{LFR}_{\max}$. The twirling residual (coherent,
non-Pauli components of the logical noise) is not captured and is
stated as a limitation; it can be probed by randomized benchmarking
style experiments, which we leave to future work.
\end{remark}

\section{Stochastic adversary: the device-aware bound}

\begin{corollary}[Device-aware stochastic-injection bound]\label{cor:stoch}
Let the adversary inject i.i.d.\ Pauli faults independently on each of
$d$ data qubits with probability $p$ in a single round, on an
implemented memory whose native fault-count distribution is
$\pi(k):=\Pr[|N|\ge k]$ (with $\pi(k)=1$ for $k\le0$). Then, by
Theorem~\ref{thm:spacetime}(c) and the law of total probability,
\[
\lambda(p)\;\le\;\sum_{w=0}^{d}\Pr\bigl[\mathrm{Bin}(d,p)=w\bigr]\,
\pi\bigl((t{+}1)-w\bigr).
\]
\end{corollary}

\begin{remark}[The naive bound is false on real devices]\label{rem:naive}
The independence form
$\lambda(p)\le\delta_0+\Pr[\mathrm{Bin}(d,p)\ge t{+}1]$
(equivalently the i.i.d.\ lower bound of the v17 ACE section with
$n{=}d$) assumes that a weight-$\le t$ injection leaves the failure
probability at its baseline $\delta_0$. Theorem~\ref{thm:spacetime}(c)
shows this is wrong whenever $\pi(1)>0$: an injection of weight $w$
\emph{amplifies} native noise, requiring only $t{+}1{-}w$ assisting
native faults. Our pre-registered campaign (below) falsified the naive
form at $p=0.10$ by $2.1$--$3.8\times$ on both sectors while the
device-aware form of Corollary~\ref{cor:stoch} held on every arm.
The correction is not cosmetic: at device-relevant parameters the
amplification term $\Pr[\mathrm{Bin}=t]\cdot\pi(1)$ dominates the
naive tail by an order of magnitude.
\end{remark}

\section{Threat model and the monitoring corollary}

\begin{definition}[Memory-integrity security notion]
The adversary wins if a logical Pauli error is applied to the stored
state without detection. Security parameter: $\lambda_{\mathcal{M}}(W)$
under $\mathfrak{A}_1^{\mathrm{st}}(W)$; a memory is
\emph{$(W,\eta)$-integrity-secure} if $\lambda_{\mathcal{M}}(W)\le\eta$.
\end{definition}

\begin{corollary}[Syndrome monitoring is insufficient]\label{cor:monitor}
The REP5 active-provenance dissociation (main text, Appendix~A) shows
that layout changes move the \emph{detection rate} while leaving the
\emph{logical} rate statistically unchanged. Hence the detection rate
is not a faithful witness of logical integrity: a memory can be
certified $(W,\eta)$-secure only by logical-level tests (e.g.\
injection probing per Theorem~\ref{thm:spacetime}), not by syndrome
statistics alone.
\end{corollary}
